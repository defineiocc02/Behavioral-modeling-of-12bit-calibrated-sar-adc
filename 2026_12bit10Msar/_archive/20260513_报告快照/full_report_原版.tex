\section{引言与设计目标}
\label{sec:intro}

\subsection{项目背景}

逐次逼近寄存器（SAR）型模数转换器因出色的能效比和数字友好的架构，在中低精度、中高速应用中占据主导地位。
传统SAR ADC面临采样开关$kT/C$噪声的根本限制——单位电容越小，$kT/C$噪声越大。
电流调节自动调零（Current-Adjusting Auto-Zeroing, CAAZ）技术在采样阶段通过预放大器负反馈环路主动抵消采样电容上的$kT/C$噪声，
在保持低功耗的同时逼近热噪声极限。

\section{噪声注入配置与CAAZ机理}
\label{sec:noise}

\subsection{Noise Factor (NF) 机制}

NF控制噪声注入功率倍数。噪声注入开关为I1（时钟链）和I14（采样开关）。

\begin{table}[htbp]
\centering
\caption{Noise Factor 配置}
\label{tab:nf}
\begin{tabular}{lcl}
\toprule
\textbf{NF} & \textbf{噪声功率倍乘} & \textbf{用途} \\
\midrule
1 & 1$\times$ & 芯片基线性能，CAAZ正常工作 \\
10 & 100$\times$ ($10^2$) & CAAZ效率表征，$\eta$计算 \\
\bottomrule
\end{tabular}
\end{table}

\subsection{CAAZ 消除效率模型}

\begin{align}
N_{total} &= N_{th} + N_s(1-\eta) \label{eq:noise} \\
\frac{N_s}{N_{th}} &= \text{NF}^2 = 100\quad(\text{NF}=10) \label{eq:nf}
\end{align}

若CAAZ完全失效，采样噪声占总噪声99\%，SNR上限仅约54.4~dB。

\begin{figure}[htbp]
\centering
\includegraphics[width=0.75\textwidth]{../../05_研究报告/figures/Fig2_CAAZ_Eta.png}
\caption{CAAZ 噪声消除效率 $\eta$ 跨批次对比。红色虚线为 85\% 目标线。}
\label{fig:eta}
\end{figure}

\begin{equation}
\eta_{26.0} = 1 - \frac{101\cdot 10^{-\Delta\text{SNR}/10} - 1}{100}
\label{eq:eta}
\end{equation}

其中 $\Delta\text{SNR} = \text{SNR}_{t_d=26\text{ns},\,\eta} - \text{SNR}_{t_d=24.05\text{ns},\,\eta\approx0}$。
推荐使用\textbf{同批次参考}以消除批次间系统偏差。

\begin{databox}{CAAZ工作时序}
CLK上升沿触发比较器 $\rightarrow$ $t_d$窗口内CAAZ预放大器建立 $\rightarrow$ COMN有效 $\rightarrow$ CCLK锁存。
$t_d$=24.05~ns时CAAZ未充分建立（$\eta\approx0$），$t_d$=26.0~ns时$\eta$显著激活。
\end{databox}

\subsection{与CAAZ的协同效应}

\begin{resultbox}{小电容 $+$ CAAZ = 论文技术亮点}

$C_u$越小$\Rightarrow C_{CDAC}$越小$\Rightarrow kT/C$越大 $\Rightarrow$ \textbf{CAAZ的SNR提升倍数越高}。
\end{resultbox}
